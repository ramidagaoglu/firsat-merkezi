🤖 7/24 Otomatik Scraper Bot (scraper.py)

Tarayıcı kapalıyken GitHub Actions sunucularının fiyat ve stok taraması yapması için repona scraper.py dosyasını ekle:

import os import json import time import requests from bs4 import BeautifulSoup DISCORD_WEBHOOK = os.getenv("DISCORD_WEBHOOK", "") PRODUCTS_FILE = "tracked_products.json" HEADERS = { "User-Agent": "Mozilla/5.0 (Linux; Android 14; SM-S928B) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Mobile Safari/537.36", "Accept-Language": "tr-TR,tr;q=0.9,en-US;q=0.8,en;q=0.7" } def send_alert(title, message): print(f"[ALARM] {title} -> {message}") if DISCORD_WEBHOOK: requests.post(DISCORD_WEBHOOK, json={"content": f"**{title}**\n{message}"}, timeout=10) def parse_price(text): cleaned = "".join(ch for ch in text if ch.isdigit() or ch == ",") cleaned = cleaned.replace(",", ".") try: return float(cleaned) except ValueError: return None def scan_all(): if not os.path.exists(PRODUCTS_FILE): print("tracked_products.json bulunamadı.") return with open(PRODUCTS_FILE, "r", encoding="utf-8") as f: products = json.load(f) for item in products: url = item.get("url") old_price = float(item.get("currentPrice", 0)) target_price = float(item.get("targetPrice", 0)) try: resp = requests.get(url, headers=HEADERS, timeout=20) if resp.status_code != 200: print(f"[UYARI] {item['title']} HTTP {resp.status_code}") continue soup = BeautifulSoup(resp.text, "html.parser") price_el = soup.select_one(".prc-dsc, [data-test-id='price-current-price'], .newPrice ins, .classifiedInfo h3") if price_el: new_price = parse_price(price_el.get_text()) if new_price and new_price != old_price: diff = new_price - old_price if diff < 0: send_alert(f"📉 FİYAT DÜŞTÜ ({item['platform']})", f"{item['title']}: {old_price} TL -> {new_price} TL") else: send_alert(f"📈 ZAM GELDİ ({item['platform']})", f"{item['title']}: {old_price} TL -> {new_price} TL") item["currentPrice"] = new_price time.sleep(2) except Exception as exc: print(f"[HATA] {url}: {exc}") with open(PRODUCTS_FILE, "w", encoding="utf-8") as f: json.dump(products, f, ensure_ascii=False, indent=2) if __name__ == "__main__": scan_all()
